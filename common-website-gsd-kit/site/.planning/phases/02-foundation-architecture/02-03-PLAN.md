  dispose(): void;                       // full three.js disposal (brief 05 bans leaks)

// src/island/systems/regressionLadder.ts (pure — BUDGETS.md §Adaptive-quality)
export type Rung = 'full' | 'dpr-down' | 'no-post' | 'no-garnish' | 'authored-cuts';
export function createLadder(cfg: { budgetMs: number }): {
  sample(nowMs: number, frameMs: number): Rung;  // feeds every frame; steps down when >20% of frames over budgetMs in a 3s window; one rung per 3s; recovers one rung per 10s under budget
};

// src/island/systems/tierMap.ts (research Code Example — verbatim semantics)
export type QualityTier = 'floor-mobile' | 'high-mobile' | 'desktop';
export function mapTier(t: { tier: number; isMobile: boolean; type: string }): QualityTier;

// src/island/systems/AssetQueue.ts (D2-20 — DI per D2-02, no module mocks)
export type Priority = 'critical' | 'soon' | 'lazy';
export class AssetQueue {
  constructor(deps: { fetchFn: typeof fetch; baseUrl: string });
  add(url: string, priority: Priority, expectedBytes: number): void;
  onProgress(fn: (fraction: number) => void): () => void;  // monotonic 0→1 over critical batch, byte-weighted; disposer-returning per conventions
  load(priority: Priority): Promise<Map<string, ArrayBuffer>>;
}
