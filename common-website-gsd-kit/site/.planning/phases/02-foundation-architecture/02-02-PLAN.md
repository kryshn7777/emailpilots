```typescript
// src/boot/gate.ts
export type BootPath = 'full' | 'static' | 'reduced';
export interface GateEnv {
  reducedMotion: boolean;   // matchMedia('(prefers-reduced-motion: reduce)').matches
  webgl2: boolean;          // !!document.createElement('canvas').getContext('webgl2')
  saveData: boolean;        // navigator.connection?.saveData === true
  override: string | null;  // URLSearchParams 'path' — allowlist-validated inside
}
export function decidePath(env: GateEnv): BootPath;

// src/island/index.ts — the ONE dynamic-import entry
export interface MountOptions {
  el: string;                              // '#stage'
  baseUrl: string;                         // import.meta.env.BASE_URL
  onProgress: (fraction: number) => void;  // drives the loader (D2-20)
  tierOverride?: string | null;            // validated '?tier=' passthrough (D2-10)
}
export function mount(opts: MountOptions): Promise<void>;
```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| URL query string → boot script | The site's only runtime input (?path, ?tier) crosses here |
| GitHub Actions → Pages deployment | CI publishes to the public origin with the ambient token |
| gwfh/Google Fonts → public/fonts | Font binaries downloaded at build-setup time, then committed |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-02-04 | Tampering (XSS-class) | ?path/?tier handling in boot script | mitigate | Literal allowlist comparison in pure decidePath / tier validation; values NEVER written to innerHTML or reflected into markup — only compared and passed as a typed enum (ASVS V5) |
| T-02-05 | Elevation of privilege | deploy-site.yml GITHUB_TOKEN | mitigate | permissions block restricted to contents:read + pages:write + id-token:write; no secrets used; workflow_dispatch ref input is operator-only (maintainers can already run arbitrary CI) — residual accepted |
| T-02-06 | Information disclosure | Static site payload | mitigate | No secrets exist in this project; workflow greps confirm zero `secrets.` references; site is fully public by design |
| T-02-07 | Tampering | Font downloads | mitigate | HTTPS-only from Google Fonts/gwfh; files committed and byte-gated (a swapped binary is caught as a size/diff anomaly in review); OFL license text kept in-repo |
| T-02-SC | Tampering | npm installs | mitigate | No new npm packages in this plan (fonts fetched as static assets precisely to keep the audited set closed); any future addition re-runs slopcheck |
</threat_model>
