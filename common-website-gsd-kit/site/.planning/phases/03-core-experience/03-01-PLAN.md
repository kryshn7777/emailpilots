// src/island/choreo/beats.ts — pure, zero imports
export interface Beat { at: number; label: string }        // at = chapter-local 0..1
export const span: (p: number, a: number, b: number) => number;   // clamped 0..1 inside [a,b]
export const CHAPTER_SPANS_VH: readonly number[];          // 7 entries, sums to 740

// src/island/choreo/cameraRig.ts — pure, plain arrays so node:test runs it
export interface CamPose { position: [number, number, number]; target: [number, number, number] }
export type CamKind = 'LOCKED' | 'SETTLE' | 'ORBIT_HOLD' | 'TILT_REVEAL' | 'PULL_WIDE';
export interface CamMove { kind: CamKind; from: CamPose; to: CamPose; span: [number, number] }
export const CHAPTER_CAMERA: readonly { start: CamPose; end: CamPose; moves: readonly CamMove[] }[];
export function poseAt(chapterIndex: number, progress: number): CamPose;

// src/island/choreo/setState.ts — the Ch(n) → Ch(n+1) handoff record
export interface SetState {
  skyId: 'sunrise' | 'noon' | 'dusk2' | 'night' | 'moonrise';
  lampOn: boolean;
  dayIndex: number;            // 1..6, the Ch4 wheel counter
  constellationCount: number;  // points currently hanging over the desk
  stack: readonly string[];    // remaining envelope tag ids, e.g. ['followup-2','followup-3']
  exposureStops: number;       // per-chapter exposure ramp offset (ART-DIRECTION)
}
export function createSetState(): SetState;
