```typescript
// src/island/choreo/beats.ts — pure, zero imports
export interface Beat { at: number; label: string }        // at = chapter-local 0..1
export const span: (p: number, a: number, b: number) => number;   // clamped 0..1 inside [a,b]
export const CHAPTER_SPANS_VH: readonly number[];          // 7 entries, sums to 740

// src/island/choreo/cameraRig.ts — pure, plain arrays so node:test runs it
export interface CamPose { position: [number, number, number]; target: [number, number, number] }
export type CamKind = 'LOCKED' | 'SETTLE' | 'ORBIT_HOLD' | 'TILT_REVEAL' | 'PULL_WIDE';
export interface CamMove { kind: CamKind; from: CamPose; to: CamPose; span: [number, number] }
export const CHAPTER_CAMERA: readonly { start: CamPose; end: CamPose; moves: readonly CamMove[] }[];
export function poseAt(chapterIndex: number, progress: number): CamPose;

// src/island/choreo/setState.ts — the Ch(n) → Ch(n+1) handoff record
export interface SetState {
  /**
   * Cumulative wheel phase — ONE continuous monotonic timeline across the whole site
   * (STORYBOARD Amendment A6). `skyId` is NOT stored: it is derived from this number by
   * `wheelSegment(phase).a` in dayWheel.ts, which owns the mapping. Storing a segment id
   * and inferring a phase from it is what produced the Ch4/Ch5 contradiction Ruling 5 fixed.
   */
  wheelPhase: number;
  lampOn: boolean;
  dayIndex: number;            // 1..6, the Ch4 wheel counter
  constellationCount: number;  // points currently hanging over the desk
  stack: readonly string[];    // remaining envelope tag ids, e.g. ['followup-2','followup-3']
  exposureStops: number;       // per-chapter exposure ramp offset (ART-DIRECTION)
  dustOpacity: number;         // 1 at the journey origin; Ch1 ramps it to 0 across 95-100% (REVIEWS.md F-4)
}
export function createSetState(): SetState;
/** Declared start/end set-state per chapter — the light/set half of the one-take contract. */
export const SET_STATE: readonly { start: SetState; end: SetState }[];   // 7 entries
```

**Ownership (this is the contract, not a suggestion — CHK-B1):** `src/island/index.ts` calls
`createSetState()` **exactly once** at mount and passes that single record into every chapter's
constructor, alongside the Room. A chapter READS it in `mount()` to know what it inherits, and its
`update()` WRITES its own end-state fields as the beats fire. No chapter calls `createSetState()`,
and no chapter keeps a private copy. `SET_STATE` is the declared table those writes must agree with,
and `setState.test.ts` asserts the seams — so a light that fails to hand over is a failing test at
wave 1, not a human eyeball finding at wave 11.
